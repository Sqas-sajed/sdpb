\section{Conventions and Notation}

\paragraph{Shifted Mandelstam variables.}
Define $\mu = s+t+u$ and
\[
  S_1 = s - \tfrac{\mu}{3},\quad S_2 = t - \tfrac{\mu}{3},\quad S_3 = u-\tfrac{\mu}{3},
\]
so $S_1+S_2+S_3=0$.

\paragraph{Crossing-symmetric basis.}
\[
  x = -(S_1 S_2+S_2 S_3+S_3 S_1),\qquad y = -S_1 S_2 S_3.
\]
Amplitude: $\mathcal{M} = \sum_{p,q\ge0} W_{p,q}\,x^p\,y^q$.

\paragraph{CSDR (n,m) index notation — UNIFIED for objectives and null constraints.}
All Wilson coefficients are labelled by a single pair $(n, m)$:
\[
  W_{n-m,\,m}:\quad
  \begin{cases}
    n \ge m \ge 0,\; 2n+m>0 & \text{objectives (includes } W_{0,m}\text{ when }n=m) \\
    n < m,\; m > n \ge 1 & \text{null constraints } (W_{n-m,m} = 0)
  \end{cases}
\]
The spectral denominator power is $\mathbf{2n+m}$ for \emph{both} cases.
Mapping to the conventional $(p,q)$ notation: $p = n-m$, $q = m$, so $2p+3q = 2n+m$.

\paragraph{Gegenbauer parameter.}
$\alpha = (d-3)/2$, giving $(2\ell+2\alpha) = (2\ell+d-3)$.

\section{Derivation of the SDPB Kernel}

\subsection{Starting point: CSDR dispersion relation (Sinha-Zahed eq.~(4))}

\[
  W_{n-m,\,m}
  = \int_{s_1>\delta_0} \frac{ds_1}{s_1}\,\Phi(s_1)
    \sum_{\ell=0,\text{even}} (2\ell+2\alpha)\,a_\ell(s_1)\,B_\ell^{(n,m)}(s_1).
\]

\subsection{Large-$s_1$ limit}

$\Phi(s_1) \to s_1^{(4-d)/2}\,\Psi(\alpha)$ with $\Psi(\alpha) > 0$.
Since $\Psi(\alpha)$ is a positive constant it does not affect the positivity
conditions; set $\Psi(\alpha)=1$.

The large-$s_1$ expansion of $B_\ell^{(n,m)}$ gives (Sinha-Zahed eq.~(11) for $m>n$):
\[
  B_\ell^{(n,m)}(s_1) \;\sim\; \frac{C_\ell^{(\alpha)}(1)}{\pi}\,
  \frac{D_{\ell,\alpha}^{(n,m)}}{s_1^{2n+m}},
\]
where $D_{\ell,\alpha}^{(n,m)}$ depends on the case (see Section~\ref{sec:D}).

\subsection{Heavy average (Extremal EFT, eq.~(1))}

\begin{equation}\label{eq:avg}
  \langle F(s_1, \ell) \rangle
  = \sum_{\ell\text{ even}} n_\ell^{(d)}\int_{\delta_0}^{\infty}\frac{ds_1}{s_1}\,
    \frac{s_1^{4-d}}{\pi}\,\rho_\ell(s_1)\,F(s_1,\ell),
\end{equation}
with $n_J^{(d)} = (4\pi)^{d/2}(d+2J-3)\Gamma(d+J-3)/\bigl(\pi\,\Gamma\!\bigl(\tfrac{d-2}{2}\bigr)\Gamma(J+1)\bigr)$.

Substituting the large-$s_1$ form of $B_\ell^{(n,m)}$ into the dispersion integral
and matching with $\langle F\rangle$, the integrand is identified as:
\[
  F(s_1,\ell) = \frac{D_{\ell,\alpha}^{(n,m)}}{n_\ell^{(d)}}\,
                C_\ell^{(\alpha)}(1)\,(2\ell+d-3)\,\frac{1}{s_1^{2n+m}}.
\]
Hence \textbf{(user notes eq.~(2))}:
\begin{equation}\label{eq:W_avg}
  \boxed{W_{n-m,\,m}
  = \Bigl\langle
      \frac{D_{\ell,\alpha}^{(n,m)}}{n_\ell^{(d)}}\,
      C_\ell^{(\alpha)}(1)\,(2\ell+d-3)\,
      \frac{1}{s_1^{2n+m}}
    \Bigr\rangle}
\end{equation}

\section{Why $n_\ell^{(d)}$ Cancels and Is Not Needed in SDPB Blocks}

Expanding $\langle F\rangle$ using the definition \eqref{eq:avg}:
\begin{align}
  \langle F\rangle
  &= \sum_\ell n_\ell^{(d)}\int\rho_\ell\,
     \frac{D_{\ell,\alpha}^{(n,m)}}{n_\ell^{(d)}}\,
     C_\ell^{(\alpha)}(1)\,(2\ell+d-3)\,\frac{1}{s_1^{2n+m}}\,(\text{measure}) \nonumber\\
  &= \sum_\ell \int\rho_\ell\,
     D_{\ell,\alpha}^{(n,m)}\,C_\ell^{(\alpha)}(1)\,(2\ell+d-3)\,
     \frac{1}{s_1^{2n+m}}\,(\text{measure}).
\end{align}
The factors $n_\ell^{(d)}$ cancel exactly.

\begin{mdframed}[backgroundcolor=green!15]
\textbf{SDPB blocks.}
The SDPB polynomial encodes the condition $\langle F\rangle\ge0$ for all
$\rho_\ell\ge0$.  Since $n_\ell^{(d)}>0$, this is equivalent to requiring
\[
  D_{\ell,\alpha}^{(n,m)}\,C_\ell^{(\alpha)}(1)\,(2\ell+d-3) \cdot (\text{polynomial in }s_1)\;\ge\;0
\]
for each even $\ell$ and $s_1\ge\delta_0$.  This is a polynomial positivity
problem in $x = s_1/\delta_0 - 1 \ge 0$, which is what SDPB enforces.
\textbf{SDPB blocks do not need $n_\ell^{(d)}$ explicitly.}
\end{mdframed}

\section{D Coefficient — Two Formulas}\label{sec:D}

\subsection{Null constraints ($m > n \ge 1$): CSDR paper eq.~(11)}

\begin{equation}\label{eq:D_null}
  D_{\ell,\alpha}^{(n,m)}
  = \Gamma(m-n)\sum_{j=n}^{m}
    \frac{(-4)^j\,(-\tfrac{\ell}{2})_j\,(\alpha+\tfrac{\ell}{2})_j\,(3j-m-2n)}
         {j!\,(m-j)!\,(j-n)!\,(\alpha+\tfrac{1}{2})_j}.
\end{equation}
This coefficient \textbf{depends on the spin $\ell$} via the Pochhammer symbols
$(-\tfrac{\ell}{2})_j$ and $(\alpha+\tfrac{\ell}{2})_j$.

Implemented in \texttt{D\_coeff(n, m, ell, alpha)} in \texttt{csdr.py}.

\begin{mdframed}[backgroundcolor=yellow!20]
Verified: $D^{(1,2)}_{\ell,\tfrac{1}{2}}$ matches the closed form
\[
  \frac{2\ell(\ell+1)\bigl(-11-5+2\ell(\ell+1)\bigr)}{2\cdot4}
\]
for $\ell = 0, 2, 4, 6, 8$ (exact rational arithmetic, checked by \texttt{check\_closed\_form\_1\_2}).
\end{mdframed}

\subsection{Objectives ($n \ge m \ge 0$): corrected formula (same parent as null)}

\textbf{Corrected formula} (user's derivation from CSDR eq.~(4), valid for $n \ge m \ge 0$):

\begin{equation}\label{eq:D_obj}
  D_{\ell,\alpha}^{(n,m)}
  = \sum_{j=0}^{m}
    \frac{(-4)^j\,(-\tfrac{\ell}{2})_j\,(\alpha+\tfrac{\ell}{2})_j\,(3j-m-2n)\,(n-j)!\,(-1)^{m+j+1}}
         {(\alpha+\tfrac{1}{2})_j\,j!\,(m-j)!\,(n-m)!}.
\end{equation}
This is obtained from the null formula \eqref{eq:D_null} by simplifying:
\[
  \frac{(3j-m-2n)\,(-n)_m}{j!\,(m-j)!\,(-n)_{j+1}}
  = \frac{(3j-m-2n)\,(n-j)!}{j!\,(m-j)!\,(n-m)!}\,(-1)^{m+j+1}
  \quad (m\le n, \; j\le m).
\]

Key features of the corrected formula:
\begin{itemize}
  \item Uses $(-\tfrac{\ell}{2})_j$ and $(\alpha+\tfrac{\ell}{2})_j$ — \textbf{same Pochhammer
        structure as the null formula \eqref{eq:D_null}}.
  \item Denominator uses $(\alpha+\tfrac{1}{2})_j$ (same as null).
  \item Factor $(n-j)!\,(-1)^{m+j+1}$ replaces $\Gamma(m-n)/(j-n)!$.
  \item Sum from $j=0$ to $m$ (not from $j=n$), because $(n-j)!$ replaces the $(j-n)!$ pole.
  \item \textbf{$D_{\ell,\alpha}^{(n,m)}$ depends on the spin $\ell$}, just like the null formula.
        The Pochhammer symbol $(-\tfrac{\ell}{2})_j$ vanishes for $j>\ell/2$ (if $\ell$ even),
        effectively truncating the sum.
  \item For $n=m$: $(n-m)! = 0! = 1$, so $W_{0,m}$ is well-defined.
\end{itemize}

Implemented in \texttt{D\_coeff\_obj(n, m, ell, alpha)} in \texttt{csdr.py}.

Selected values for $\alpha=\tfrac{1}{2}$ (i.e.\ $d=4$), $\ell=0$ (only $j=0$ survives since $(-0)_j=0$ for $j\ge1$):
\begin{align*}
  D^{(1,0)}_{\ell=0} &= 2, & D^{(2,0)}_{\ell=0} &= 4, & D^{(1,1)}_{\ell=0} &= -3.
\end{align*}
For $\ell=2$:
\begin{align*}
  D^{(2,1)}_{\ell=2} &= 2, & D^{(2,1)}_{\ell=0} &= -10.
\end{align*}
Verified by the unit check \texttt{check\_obj\_D\_basic()}.

\section{SDPB Polynomial Matrix Program Structure}

\subsection{Decision variables (unified (n,m) notation)}

\begin{itemize}
  \item $z_{n,m}$: objective Wilson coefficients, $(n,m)$ with $n\ge m\ge0$ and $0<2n+m\le K$.
        Includes $W_{0,m}$ (i.e.\ $n=m\ge1$) as physically valid operators.
  \item $c_{n,m}$: null constraint Lagrange multipliers, $(n,m)$ with $m>n\ge1$ and $2n+m\le K$.
\end{itemize}
Both sets use the same (n,m) index convention as $W_{n-m,m}$.

\subsection{Spin block for spin $\ell$}

Setting $s_1 = \delta_0(1+x)$ so $x\ge0$, the polynomial block is
\begin{equation}\label{eq:block}
  P^{(\ell)}(x)
  = \sum_{(n,m)\in\text{obj}} z_{n,m}\,\kappa_\ell^{\rm obj}(n,m)\,(1+x)^{K-2n-m}
  + \sum_{(n,m)\in\text{null}} c_{n,m}\,\kappa_\ell^{\rm null}(n,m)\,(1+x)^{K-2n-m},
\end{equation}
where
\begin{align}
  \kappa_\ell^{\rm obj}(n,m) &= D_{\ell,\alpha}^{(n,m)}\cdot C_\ell^{(\alpha)}(1)\cdot(2\ell+d-3),
  & \text{(from eq.~\eqref{eq:D_obj}, }\ell\text{-dependent }D\text{)} \label{eq:kappa_obj}\\
  \kappa_\ell^{\rm null}(n,m) &= D_{\ell,\alpha}^{(n,m)}\cdot C_\ell^{(\alpha)}(1)\cdot(2\ell+d-3).
  & \text{(from eq.~\eqref{eq:D_null}, }\ell\text{-dependent }D\text{)} \label{eq:kappa_null}
\end{align}

\subsection{DampedRational prefactor}

The SDPB PMP JSON format uses the key \texttt{"DampedRational"} (not \texttt{"prefactor"})
inside each block.  The prefactor is $e^{-x}/(1+x)^K$, encoded as:
\begin{verbatim}
"DampedRational": {
  "base": "0.36787944...",   // e^{-1}
  "constant": "1",
  "poles": ["-1", ..., "-1"] // K poles at x = -1
}
\end{verbatim}

\subsection{SDPB optimization}

\begin{alignat}{2}
  &\text{maximize}  &&\quad z_{n_0,m_0} \nonumber\\
  &\text{subject to}&&\quad z_{n_1,m_1} = 1 \quad(\text{normalization}) \nonumber\\
  &                 &&\quad P^{(\ell)}(x)\ge0\quad\forall\,\ell\text{ even}\ge0,\;x\ge0.
\end{alignat}

\section{PMP File Construction (from SDPB README)}

The SDPB solver takes a JSON file following the \texttt{PositiveMatrixWithPrefactor}
format.  The top-level structure is:

\begin{verbatim}
{
  "objective": [b_0, b_1, ..., b_{N-1}],
  "normalization": [c_0, c_1, ..., c_{N-1}],
  "PositiveMatrixWithPrefactorArray": [
    // one block per spin ell = 0, 2, 4, ..., ell_max
    {
      "DampedRational": {
        "base": "<e^{-1} to high precision>",
        "constant": "1",
        "poles": ["-1", "-1", ..., "-1"]   // K entries
      },
      "polynomials": [
        [
          // rows of the 1x1 matrix (one polynomial per decision variable)
          [ [c_0, c_1, ..., c_K] ]   // var 0
          ...
        ]
      ]
    },
    ...
  ]
}
\end{verbatim}

Produced by \texttt{generate\_csdr\_pmp(obj\_index, norm\_index, d, K, max\_spin, delta0, precision)}.
Parameters:
\begin{itemize}
  \item \texttt{obj\_index}: $(n,m)$ of the Wilson coefficient to bound ($n\ge m\ge0$).
  \item \texttt{norm\_index}: $(n,m)$ of the Wilson coefficient to normalize to 1.
  \item \texttt{d}: spacetime dimension (required).
  \item \texttt{K}: maximum spectral power $2n+m$ (default 8).
  \item \texttt{max\_spin}: maximum even spin $\ell$ included (default 10).
  \item \texttt{delta0}: IR cutoff; integration starts at $s_1=\delta_0$ (default 1).
  \item \texttt{precision}: decimal precision for output coefficients (default 200).
\end{itemize}
