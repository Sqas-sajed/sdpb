\section{Conventions}

\paragraph{Shifted Mandelstam variables.}
Let $\mu = s+t+u = 4m^2$.  Define
\[
  S_1 = s - \tfrac{\mu}{3},\quad S_2 = t - \tfrac{\mu}{3},\quad S_3 = u-\tfrac{\mu}{3},
\]
so $S_1+S_2+S_3=0$.  This is \textbf{not} a choice: $S_i$ are the centred
Mandelstam variables used throughout the CSDR paper.

\paragraph{Crossing-symmetric basis.}
Because $S_1+S_2+S_3=0$ the natural crossing-symmetric invariants are
\[
  x = -(S_1 S_2+S_2 S_3+S_3 S_1),\qquad y = -S_1 S_2 S_3.
\]
The amplitude is expanded as $\mathcal{M} = \sum_{p,q\ge0} W_{p,q}\,x^p\,y^q$.

\paragraph{CSDR index notation.}
The CSDR paper writes $W_{n-m,m}$ with first index $= n-m$ and second $= m$:
\begin{itemize}
  \item $n-m\ge0$ (i.e.\ $n\ge m$): \textbf{objectives} ($W_{p,q}\ge0$ from unitarity).
  \item $n-m<0$ (i.e.\ $n<m$, equivalently $m>n\ge1$): \textbf{null constraints} ($W_{n-m,m}=0$).
\end{itemize}
The power of $s_1$ in the kernel denominator is $2n+m = 2(n-m)+3m = 2p+3q$.

\paragraph{Spacetime dimension.}
$d$ is a free parameter.  The Gegenbauer parameter is $\alpha=(d-3)/2$, giving
$(2\ell+2\alpha)=(2\ell+d-3)$.

\section{Derivation of the SDPB Kernel — Step-by-step Record}

This section gives the complete derivation used in the code, so it can
be audited line by line.

\subsection{Step 1: CSDR dispersion relation (Sinha-Zahed eq.~(4))}

\[
  W_{n-m,\,m}
  = \int_{s_1>\delta_0} \frac{ds_1}{s_1}\,\Phi(s_1)
    \sum_{\ell=0,\text{even}}^\infty (2\ell+2\alpha)\,a_\ell(s_1)\,B_\ell^{(n,m)}(s_1).
\]

\subsection{Step 2: Large-$s_1$ limit of $\Phi$}

\[
  \Phi(s_1) = \frac{\sqrt{s_1+M/3}}{(s_1-2M/3)^\alpha}\,\Psi(\alpha)
  \;\xrightarrow{s_1\to\infty}\; s_1^{1/2-\alpha}\,\Psi(\alpha).
\]
Since $\alpha=(d-3)/2$, the exponent is $\tfrac{1}{2}-\tfrac{d-3}{2}=\tfrac{4-d}{2}$.
Therefore $\Phi\sim s_1^{(4-d)/2}\,\Psi(\alpha)$.  Because $\Psi(\alpha)>0$ is just
a positive constant, we set $\Psi(\alpha)=1$.

\subsection{Step 3: Large-$s_1$ limit of the kernel $B_\ell^{(n,m)}$}

For $m>n\ge1$ (null constraints), CSDR eq.~(11) gives:
\[
  B_\ell^{(n,m)}(s_1)\;\sim\;\frac{C_\ell^{(\alpha)}(1)}{\pi}\,\frac{D_{\ell,\alpha}^{(n,m)}}{s_1^{2n+m}},
\]
where
\[
  D_{\ell,\alpha}^{(n,m)}
  = \Gamma(m-n)\sum_{j=n}^{m}
    \frac{(-4)^j\,\bigl(-\tfrac{\ell}{2}\bigr)_j\,\bigl(\alpha+\tfrac{\ell}{2}\bigr)_j
          \,(3j-m-2n)}
         {j!\,(m-j)!\,(j-n)!\,\bigl(\alpha+\tfrac{1}{2}\bigr)_j}.
\]

\begin{mdframed}[backgroundcolor=yellow!20]
\textbf{Code verification.}
The function \texttt{D\_coeff(n,m,ell,alpha)} in \texttt{csdr.py} evaluates
this sum exactly using Python's \texttt{Fraction} arithmetic.
The test \texttt{check\_closed\_form\_1\_2()} verifies $(n,m)=(1,2)$ against
the closed form
$D^{(1,2)}=2\ell(\ell+2\alpha)(-11-10\alpha+2\ell(\ell+2\alpha))/((2\alpha+1)(2\alpha+3))$
for $\ell=0,2,4,6,8$.  All 5 cases pass (confirmed by running the code).
\end{mdframed}

\textbf{Note on the D formula vs.\ user notes.}  The user's LaTeX notes write
$(-\tfrac{n}{2})_j$ and $(\alpha-\tfrac{1}{2})_j$ in the Pochhammer symbols.
Based on the CSDR paper and the verified closed-form test, the correct symbols
are $(-\tfrac{\ell}{2})_j$ and $(\alpha+\tfrac{\ell}{2})_j$.  The user's notation
appears to contain typos ($n\to\ell$ and $-\tfrac12\to+\tfrac{\ell}{2}$).

\subsection{Step 4: Heavy average}

The Extremal EFT heavy average is (user notes eq.~(1)):
\[
  \langle F(m^2,J)\rangle
  = \sum_{J\text{ even}} n_J^{(d)} \int_{M^2}^{\infty}\frac{dm^2}{m^2}\,
    \frac{m^{4-d}}{\pi}\,\rho_J(m^2)\,F(m^2,J),
\]
with
\[
  n_J^{(d)}
  = \frac{(4\pi)^{d/2}(d+2J-3)\,\Gamma(d+J-3)}{\pi\,\Gamma\!\bigl(\tfrac{d-2}{2}\bigr)\,\Gamma(J+1)}.
\]

Substituting the dispersion result, the integrand for $W_{n-m,m}$ is proportional to
\[
  n_\ell^{(d)}\cdot\rho_\ell\cdot F(s_1,\ell),\qquad
  F(s_1,\ell) = \frac{D_{\ell}^{(n,m)}}{n_\ell^{(d)}}\,C_\ell^{(\alpha)}(1)\,(2\ell+d-3)\,\frac{1}{s_1^{2n+m}}.
\]

\subsection{Step 5: Cancellation of $n_\ell^{(d)}$ — explicit proof}

Expanding the heavy average:
\begin{align}
  \langle F\rangle
  &= \sum_\ell n_\ell^{(d)}
     \int \rho_\ell\,
     \frac{D_\ell^{(n,m)}}{n_\ell^{(d)}}\,
     C_\ell^{(\alpha)}(1)\,(2\ell+d-3)\,\frac{1}{s_1^{2n+m}}\,
     (\text{measure}) \nonumber\\
  &= \sum_\ell \int \rho_\ell\,
     D_\ell^{(n,m)}\,C_\ell^{(\alpha)}(1)\,(2\ell+d-3)\,\frac{1}{s_1^{2n+m}}\,
     (\text{measure}).
\end{align}
The factor $n_\ell^{(d)}$ cancels exactly.

\begin{mdframed}[backgroundcolor=green!15]
\textbf{Why SDPB does not need $n_\ell^{(d)}$.}
For SDPB we need: can we certify $\langle F\rangle\ge0$ for all $\rho_\ell\ge0$?
Since $n_\ell^{(d)}>0$, the condition $\langle F\rangle\ge0$ for all $\rho_\ell\ge0$
is equivalent to requiring the integrand $D_\ell^{(n,m)}\cdot C_\ell^{(\alpha)}(1)\cdot
(2\ell+d-3)/s_1^{2n+m}$ to be non-negative for each $\ell$ and $s_1$.
This is a polynomial positivity condition on the variable $x=s_1/\delta_0-1\ge0$,
which is what SDPB solves.  Dividing out the positive constant $n_\ell^{(d)}$
does not change the feasibility region.

Therefore:
\[\boxed{\text{SDPB block for spin }\ell: \quad \kappa_\ell^{(n,m)} = D_\ell^{(n,m)}\cdot C_\ell^{(\alpha)}(1)\cdot (2\ell+d-3).}\]
\end{mdframed}

\subsection{Result: User notes eq.~(2)}

\begin{equation}\label{eq:W_kernel}
  W_{n-m,\,m}
  = \Bigl\langle
      \frac{D_{\ell,\alpha}^{(n,m)}\,C_\ell^{(\alpha)}(1)\,(2\ell+d-3)}{n_\ell^{(d)}\,s_1^{2n+m}}
    \Bigr\rangle
  \quad(m>n\ge1,\;\text{null constraints})
\end{equation}
In the SDPB polynomial, the $1/n_\ell^{(d)}$ cancels with the measure, leaving
$\kappa_\ell = D\cdot C_\ell(1)\cdot(2\ell+d-3)$ as the per-spin coefficient.

\section{SDPB Polynomial Matrix Program Structure}

\subsection{Decision variables}

\begin{itemize}
  \item $z_{p,q}$: for each objective pair $(p,q)$ with $p\ge1$, $q\ge0$, $2p+3q\le K$.
  \item $c_{n,m}$: for each null constraint pair $(n,m)$ with $m>n\ge1$, $2n+m\le K$.
\end{itemize}
Total: $N_{\rm obj}+N_{\rm null}$ decision variables.

\subsection{Spin block for spin $\ell$}

Setting $s_1 = \delta_0(1+x)$ (so $x\ge0$ at threshold), the per-spin polynomial is
\begin{align}
  P^{(\ell)}(x)
  &= \sum_{(p,q)}\!z_{p,q}\,\kappa_\ell^{\rm obj}(p,q)\,(1+x)^{K-2p-3q} \nonumber\\
  &\quad+ \sum_{(n,m)}\!c_{n,m}\,\kappa_\ell^{\rm null}(n,m)\,(1+x)^{K-2n-m}.
\end{align}
Prefactor: $e^{-x}/(1+x)^K$ (DampedRational with pole $-1$ of multiplicity $K$).

\subsection{PMP JSON format (SDPB requirement)}

The SDPB PMP JSON format uses the key \texttt{"DampedRational"} (not
\texttt{"prefactor"}) inside each block.  Verified against the canonical example
in \texttt{test/data/end-to-end\_tests/1d/input/pmp.json}.

\begin{verbatim}
{
  "objective": [...],
  "normalization": [...],
  "PositiveMatrixWithPrefactorArray": [
    {
      "DampedRational": {
        "base": "0.36787...",   // e^{-1}
        "constant": "1",
        "poles": ["-1", ..., "-1"]   // K poles at -1
      },
      "polynomials": [[ [[c_0, c_1, ..., c_K]] ]]
    },
    ...
  ]
}
\end{verbatim}

\subsection{SDPB optimization}

\begin{alignat}{2}
  &\text{maximize}  &&\quad z_{p_0,q_0} \nonumber\\
  &\text{subject to}&&\quad z_{p_1,q_1} = 1 \quad\text{(normalization)}\nonumber\\
  &                 &&\quad P^{(\ell)}(x)\ge0\quad\forall\,\ell\in\{0,2,\ldots\},\,\forall\, x\ge0.
\end{alignat}
